fn first(xs: [int]): Option[int] {
    if len(xs) == 0 {
        return Option::None;
    }
    return Option::Some(xs[0]);
}

fn double_first(xs: [int]): Option[int] {
    let x: int = first(xs)?;
    return Option::Some(x * 2);
}
