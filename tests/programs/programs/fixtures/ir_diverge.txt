// IR fixture: early returns from loops and arms, code after diverging calls (pruned), an if expression whose branches
// both diverge, a never match expression, a never function, and for/if/while/match/&&/index-assign whose bounds,
// conditions, scrutinees and operands are never calls (wrapped in identity functions), each followed by more code.
fn die(msg: string): never {
    eprint(msg);
    exit(3);
}

fn id_int(x: int): int {
    return x;
}

fn id_str(s: string): string {
    return s;
}

fn id_bool(b: bool): bool {
    return b;
}

fn find(xs: [int], want: int): int {
    for i in 0..len(xs) {
        if xs[i] == want {
            return i;
        }
    }
    return -1;
}

fn pick(o: Option[int]): int {
    let Option::Some(v) = o else {
        return 0;
    };
    if let Option::Some(w) = o {
        return v + w;
    }
    return v;
}

fn never_if(b: bool): int {
    let x: int = if b { die("a") } else { die("b") };
    return x;
}

fn never_match(o: Option[int]): int {
    let y: int = match o {
        Option::Some(_) => die("some"),
        Option::None => {
            panic("none");
        }
    };
    return y;
}

fn after(flag: bool, xs: [int]): int {
    if flag {
        for k in 0..id_int(die("r")) {
            print(k);
        }
    }
    if flag {
        if id_bool(die("b")) {
            print("then");
        } else {
            print("else");
        }
    }
    if flag {
        match id_str(die("s")) {
            "zz_unreachable" | "yy" => print("a"),
            "xx" => print("c"),
            _ => print("d"),
        }
    }
    if flag {
        match id_int(die("i")) {
            0 => print("zero"),
            _ => print("other"),
        }
    }
    if flag {
        while id_bool(die("w")) {
            print("w");
        }
    }
    if flag {
        let z: bool = id_bool(die("and")) && id_bool(die("rhs"));
        xs[id_int(die("idx"))] += id_int(die("val"));
        print(z);
    }
    return 1;
}

fn main(): int {
    print(find([1, 2, 3], 2) + pick(Option::Some(4)));
    if false {
        print(never_if(true) + never_match(Option::None) + after(true, [1]));
    }
    return 0;
}
