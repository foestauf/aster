# Aster v0.3: Input and Program Arguments Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let Aster programs read files and stdin and receive command-line arguments, and prove it with an Aster lexer written in Aster.

**Architecture:** `read_file` returns a predeclared enum `ReadResult { Ok(string), Err(string) }`. The checker seeds it into its enum table, and it reaches the IR and C only when a program mentions it. `read_file` lowers to a dedicated IR instruction `read_file ok, text, path` (C: `text = aster_rt_read_file(path, &ok)`), then a `br` and two ordinary `enum_new`s, so the runtime never needs to know enum layouts. `read_stdin` is an ordinary signature builtin. `fn main(args: [string]): int` switches the C wrapper to `main(argc, argv)`.

**Tech Stack:** TypeScript (Node 24, pnpm, vitest, oxlint), C11 runtime compiled with the system `cc`.

**Spec:** `docs/superpowers/specs/2026-10-02-aster-v0.3-design.md`

## Global Constraints

- Generated C and the runtime must compile cleanly under `-std=c11 -Wall -Werror` (the golden suite builds with `-Werror`).
- Existing golden programs and unit tests keep passing. The only existing expectations allowed to change are the `main` signature message (checker unit test) and the `parseExpectations` result shape (harness unit test).
- The `main` error message is exactly `'main' must have signature 'fn main(): int' or 'fn main(args: [string]): int'`.
- The redefinition message is exactly `'ReadResult' is a builtin type and cannot be redefined`.
- `read_file` failure messages are `<path>: <strerror text>`, and `<path>: invalid path` for a path containing `\0`, with the path printed up to its first `\0`.
- The `read_stdin` panic is `cannot read stdin: <strerror text>`.
- Conventional commits, each ending with the line `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- Before each commit, run `pnpm typecheck && pnpm lint && pnpm test`. All three must pass.

## Review Focus

1. **Files larger than the runtime's initial buffer.** A 10,000-byte file must read completely. This is pinned by `fixtures/big.txt` in Task 5.
2. **Binary content.** A file containing a `\0` byte must keep its full length, not be truncated at the NUL. This is pinned by `fixtures/nul.bin` in Task 5.
3. **Program arguments that look like options.** `aster run f.aster -- a -b --` must pass `-b` and the second `--` through untouched. This is pinned in the Task 6 CLI test.
4. **Non-ASCII arguments, stdin and source.** These must stay as raw bytes, with `len` counting bytes. This is pinned by `é` in `io/args.aster` (Task 2), `wörld` in `io/stdin.aster` (Task 3), and the non-ASCII lexer fixtures (Task 7).
5. **A BOM or other unusual bytes at the start of a source file.** The lexer comparison must not silently strip a BOM. Pinned by decoding with `ignoreBOM: true` in Task 7's test.

```ts
export interface Expectations {
  stdout: string;
  stderr: string;
  exitCode: number;
  /** `line:col message` entries; non-empty means compile-only. */
  errors: string[];
  /** Program arguments, split on whitespace. */
  args: string[];
  /** Text fed to the program's stdin; empty when there is no `expect-stdin` block. */
  stdin: string;
}

const isComment = (line: string): boolean => line.startsWith('//');
const isDirective = (line: string): boolean => /^\/\/ ?expect-/.test(line);
const commentBody = (line: string): string => line.replace(/^\/\/ ?/, '');

/** Reads `// expect-*` directives from the leading comment block of a golden program. */
export function parseExpectations(text: string): Expectations {
  const lines = text.split(/\r?\n/);
  const result: Expectations = { stdout: '', stderr: '', exitCode: 0, errors: [], args: [], stdin: '' };
  let i = 0;
  /** Reads the comment lines after a block directive, each terminated by a newline. */
  const readBlock = (): string => {
    let block = '';
    while (i < lines.length && isComment(lines[i]) && !isDirective(lines[i])) {
      block += `${commentBody(lines[i])}\n`;
      i++;
    }
    return block;
  };
  while (i < lines.length && isComment(lines[i])) {
    const body = commentBody(lines[i]);
    i++;
    if (body === 'expect-stdout:') {
      result.stdout += readBlock();
      continue;
    }
    if (body === 'expect-stdin:') {
      result.stdin += readBlock();
      continue;
    }
    const exit = /^expect-exit: (-?\d+)$/.exec(body);
    const stderr = /^expect-stderr: (.*)$/.exec(body);
    const error = /^expect-error: (.*)$/.exec(body);
    const args = /^expect-args:(.*)$/.exec(body);
    if (exit) result.exitCode = Number(exit[1]);
    else if (stderr) result.stderr += `${stderr[1]}\n`;
    else if (error) result.errors.push(error[1]);
    else if (args) result.args = args[1].split(/\s+/).filter((a) => a.length > 0);
    // A typo'd directive would otherwise silently fall back to a default and weaken the test.
    else if (body.startsWith('expect-')) throw new Error(`unrecognised directive '${lines[i - 1]}'`);
  }
  return result;
}
```

/*
 * Reads `f` to EOF into a buffer that doubles as it fills, so it works whether or not the stream's size is known.
 * Sets *ok to false when the stream reports an error; errno then holds the reason.
 */
static aster_string read_all(FILE *f, bool *ok) {
    int64_t cap = 4096;
    int64_t len = 0;
    char *buf = alloc_bytes(cap);
    for (;;) {
        len += (int64_t)fread(buf + len, 1, (size_t)(cap - len), f);
        if (len < cap) break; /* a short read means EOF or an error */
        cap *= 2;
        buf = realloc(buf, (size_t)cap);
        if (buf == NULL) aster_rt_panic_cstr("out of memory");
    }
    *ok = !ferror(f);
    aster_string s = { buf, len };
    return s;
}
```

aster_string aster_rt_read_stdin(void) {

    aster_string s = read_all(stdin, &ok);

        snprintf(msg, sizeof msg, "cannot read stdin: %s", strerror(errno));

static aster_string path_error(aster_string path, int64_t path_len, const char *reason) {
    int64_t reason_len = (int64_t)strlen(reason);
    int64_t len = path_len + 2 + reason_len;
    char *buf = alloc_bytes(len);
    if (path_len > 0) memcpy(buf, path.ptr, (size_t)path_len);
    memcpy(buf + path_len, ": ", 2);
    memcpy(buf + path_len + 2, reason, (size_t)reason_len);
    aster_string s = { buf, len };
    return s;
}

aster_string aster_rt_read_file(aster_string path, bool *ok) {
    const char *nul = path.len > 0 ? memchr(path.ptr, '\0', (size_t)path.len) : NULL;
    if (nul != NULL) {
        *ok = false;
        return path_error(path, (int64_t)(nul - path.ptr), "invalid path");
    }
    char *cpath = alloc_bytes(path.len + 1);
    if (path.len > 0) memcpy(cpath, path.ptr, (size_t)path.len);
    cpath[path.len] = '\0';
    FILE *f = fopen(cpath, "rb");
    free(cpath);
    if (f == NULL) {
        *ok = false;
        return path_error(path, path.len, strerror(errno));
    }
    errno = 0;
    aster_string contents = read_all(f, ok);
    int err = errno;
    fclose(f);
    if (!*ok) return path_error(path, path.len, strerror(err));
    return contents;

// `<kind> <start> <end> <text>`, ending with `eof <n> <n>`, then one line per lexical error,

// expect-stdout:
// fn 0 2 fn
// ident 3 7 main

// ) 8 9 )
// : 9 10 :
// ident 11 14 int
// { 15 16 {

// ident 25 26 s
// : 26 27 :
// ident 28 34 string

struct Token {

struct LexError {

    end: int,

fn min(a: int, b: int): int {

fn is_digit(c: int): bool {

    return (c >= 97 && c <= 122) || (c >= 65 && c <= 90) || c == 95;

// Byte length of the UTF-8 sequence that starts with byte c.
fn utf8_len(c: int): int {
    if c >= 240 {
        return 4;
    }
    if c >= 224 {
        return 3;
    }
    if c >= 192 {
        return 2;
    }
    return 1;

fn contains(words: [string], w: string): bool {
    for x in words {

        }
    }

fn peek(lx: Lexer, offset: int): int {
    let i: int = lx.pos + offset;
    if i < len(lx.src) {
        return byte_at(lx.src, i);
    }
    return -1;
}

// Adds a token that runs from `start` to the current position.
fn push_token(lx: Lexer, kind: string, start: int) {
    push(lx.tokens, Token { kind: kind, start: start, end: lx.pos });
}

fn add_error(lx: Lexer, start: int, end: int, message: string) {
    push(lx.errors, LexError { start: start, end: end, message: message });
}

// Lexes a string literal starting at the opening quote. A literal ends at the closing quote, a newline or the end of
// the source; only the first counts as terminated.
fn lex_string(lx: Lexer) {
    let src: string = lx.src;
    let start: int = lx.pos;
    lx.pos += 1;
    var terminated: bool = false;
    while lx.pos < len(src) && byte_at(src, lx.pos) != 10 {
        let c: int = byte_at(src, lx.pos);
        if c == 34 {
            lx.pos += 1;
            terminated = true;
            break;
        }
        if c != 92 {
            lx.pos += 1;
            continue;
        }
        let next: int = peek(lx, 1);
        if is_escape(next) {
            lx.pos += 2;
            continue;
        }
        var width: int = 1;
        if next != -1 && next != 10 {
            width = 1 + utf8_len(next);
        }
        let end: int = min(lx.pos + width, len(src));
        add_error(lx, lx.pos, end, "invalid escape sequence '" + substring(src, lx.pos, end) + "'");
        lx.pos = end;
    }
    if !terminated {
        add_error(lx, start, lx.pos, "unterminated string literal");
    }
    push_token(lx, "string", start);
}

        let start: int = lx.pos;
        let c: int = byte_at(src, start);

            lx.pos += 1;
        } else if c == 47 && peek(lx, 1) == 47 {
            while lx.pos < len(src) && byte_at(src, lx.pos) != 10 {
                lx.pos += 1;
            }
        } else if is_digit(c) {
            while lx.pos < len(src) && is_digit(byte_at(src, lx.pos)) {
                lx.pos += 1;
            }
            push_token(lx, "int", start);
        } else if is_ident_start(c) {
            while lx.pos < len(src) && is_ident_part(byte_at(src, lx.pos)) {
                lx.pos += 1;
            }
            let word: string = substring(src, start, lx.pos);
            if word == "_" || is_keyword(word) {
                push_token(lx, word, start);
            } else {
                push_token(lx, "ident", start);
            }
        } else if c == 34 {
            lex_string(lx);
        } else if start + 2 <= len(src) && is_punct(substring(src, start, start + 2)) {
            lx.pos += 2;
            push_token(lx, substring(src, start, lx.pos), start);
        } else if is_punct(substring(src, start, start + 1)) {
            lx.pos += 1;
            push_token(lx, substring(src, start, lx.pos), start);
        } else {
            lx.pos = min(start + utf8_len(c), len(src));
            add_error(lx, start, lx.pos, "unexpected character '" + substring(src, start, lx.pos) + "'");
        }
    }
    push_token(lx, "eof", lx.pos);
    return lx;
}

// Prints every token, then every error, one per line. Returns 1 if there were errors.
fn report(lx: Lexer): int {
    for t in lx.tokens {
        let span: string = int_to_string(t.start) + " " + int_to_string(t.end);
        if t.kind == "eof" {
            print("eof " + span);
        } else {
            print(t.kind + " " + span + " " + substring(lx.src, t.start, t.end));
        }
    }
    for e in lx.errors {
        print("error " + int_to_string(e.start) + " " + int_to_string(e.end) + " " + e.message);
    }
    if len(lx.errors) > 0 {
        return 1;
    }
    return 0;
}

fn main(args: [string]): int {
    if len(args) != 1 {
        print("usage: lex <file>");
        return 2;
    }
    match read_file(args[0]) {
        ReadResult::Ok(src) => {
            return report(lex(src));
        }
        ReadResult::Err(msg) => {
            print(msg);
            return 2;
        }
    }
}
```
